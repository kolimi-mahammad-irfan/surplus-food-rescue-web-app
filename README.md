function expired(i){return i.status==='Available'&&(i.date<today()||(i.date===today()&&i.time<new Date().toTimeString().slice(0,5)))}
function status(i){return expired(i)?'Expired':i.status}
function fmt(d){return d?new Date(d+'T00:00:00').toLocaleDateString(undefined,{day:'numeric',month:'short',year:'numeric'}):'—'}
function card(i){let st=status(i),ico={'Cooked meals':'♨','Bakery':'♧','Produce':'❋','Packaged food':'▣','Other':'＋'}[i.type]||'♧',tag=st==='Claimed'?'claimed':st==='Expired'?'warn':'';let actions='';if(st==='Available'&&$('role').value!=='donor')actions+=`<button class="btn primary sm" onclick="claim('${i.id}')">Claim donation</button>`;if(st==='Available'&&$('role').value==='donor')actions+=`<button class="btn sm" onclick="edit('${i.id}')">Edit</button><button class="btn danger sm" onclick="removeItem('${i.id}')">Delete</button>`;if(st==='Claimed')actions+=`<button class="btn sm" onclick="collect('${i.id}')">Confirm collection</button>`;return `<article class="foodcard"><div class="foodphoto">${ico}</div><div class="foodinfo"><div class="foodtitle"><div><h3>${esc(i.name)}</h3><div class="muted" style="font-size:11px">${esc(i.donor)}</div></div><span class="tag ${tag}">${esc(st)}</span></div><div class="meta"><span>♧ ${Number(i.portions)} portions / units</span><span>⌖ ${esc(i.address)}</span><span>◷ ${esc(fmt(i.date))} · ${esc(i.time)}</span></div>${i.ingredients?`<p class="desc"><b>Ingredients / allergens:</b> ${esc(i.ingredients)}</p>`:''}${i.notes?`<p class="desc">${esc(i.notes)}</p>`:''}<div class="actions">${actions}</div></div></article>`}

Use “Post a donation” to add one.</div>'}
function refresh(){let available=data.listings.filter(i=>status(i)==='Available');$('sAvailable').textContent=available.length;$('sPortions').textContent=data.listings.reduce((s,i)=>s+Number(i.portions||0),0).toLocaleString();$('sPicked').textContent=data.listings.filter(i=>i.status==='Collected').length;$('sClaims').textContent=data.listings.filter(i=>i.status==='Claimed').length;renderListings('homeList',$('homeSearch').value);renderListings('allList',$('listSearch')?.value||'');renderActivity();renderClaims();renderReports()}
function renderActivity(){let el=$('activity');el.innerHTML=data.activity.length?data.activity.slice(0,5).map(a=>`<div class="act"><div class="acticon">↻</div><div><strong>${esc(a.title)}</strong><p>${esc(a.detail)} · ${new Date(a.at).toLocaleString()}</p></div></div>`).join(''):'<div class="empty">No activity yet. function claim(id){let i=data.listings.find(x=>x.id===id);if(!i||status(i)!=='Available'){toast('This listing is no longer available.');refresh();return}let who=prompt('Enter your NGO or recipient organisation name:');if(!who||!who.trim())return;i.status='Claimed';i.claimedBy=who.trim();i.claimedAt=new Date().toISOString();log('Donation claimed',`${i.name} claimed by ${i.claimedBy}`);save();refresh();toast('Claim recorded locally. function collect(id){let i=data.listings.find(x=>x.id===id);if(!i||i.status!=='Claimed'){toast('Only claimed donations can be marked collected.');return}if(!confirm(`Confirm "${i.name}" was handed over and collected?`))return;i.status='Collected';i.collectedAt=new Date().toISOString();log('Food handover confirmed',`${i.name} · ${i.portions} portions`);save();refresh();toast('Collection recorded.')}
function removeItem(id){let i=data.listings.find(x=>x.id===id);if(!i||i.status!=='Available')return;if(!confirm(`Delete listing "${i.name}"?`))return;data.listings=data.listings.filter(x=>x.id!==id);log('Listing deleted',i.name);save();refresh();toast('Listing deleted.')}
function edit(id){let i=data.listings.find(x=>x.id===id);if(!i||i.status!=='Available')return;let n=prompt('Food name:',i.name);if(n===null)return;let p=prompt('Number of portions / units:',String(i.portions));if(p===null)return;let a=prompt('Pickup address:',i.address);if(a===null)return;if(!n.trim()||!Number.isInteger(Number(p))||Number(p)<1||!a.trim()){toast('Enter a valid name, quantity and address.');return}i.name=n.trim();i.portions=Number(p);i.address=a.trim();log('Listing updated',i.name);save();refresh();toast('Listing updated.')}
